# Farm-Layer Scaffold Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up a walking-skeleton scaffold for the True Beef farm layer: an Expo/React Native app talking to a FastAPI/Postgres backend whose core is a certification-agnostic, append-only, hash-chained animal ledger — with one end-to-end path proven working and tamper-detection tested.

**Architecture:** Monorepo. The Python/FastAPI API owns the ledger schema and is the single source of truth; the mobile app is generated against the API's OpenAPI (`make gen-client`). The ledger is a single append-only `ledger_events` table where each row hash-chains to the previous one for its animal; a Postgres trigger blocks `UPDATE`/`DELETE`. Certifier attestations are just ledger events carrying a signature — no separate table.

**Tech Stack:** Python 3.11+, FastAPI, SQLAlchemy 2.0, Alembic, Postgres 16 (docker-compose), pytest. Expo / React Native (TypeScript), expo-router, expo-sqlite, `openapi-typescript`. GitHub Actions CI.

```python
import hashlib
import json

GENESIS_HASH = "0" * 64


def canonical_json(payload: dict) -> str:
    """Stable, key-order-independent JSON so hashes are reproducible."""
    return json.dumps(payload, sort_keys=True, separators=(",", ":"), ensure_ascii=False)


def compute_hash(
    animal_id: str, seq: int, event_type: str, payload: dict, prev_hash: str
) -> str:
    material = f"{animal_id}|{seq}|{event_type}|{canonical_json(payload)}|{prev_hash}"
    return hashlib.sha256(material.encode("utf-8")).hexdigest()
```

```python
import uuid
from datetime import date

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.hashchain import GENESIS_HASH, compute_hash
from app.models import Animal, LedgerEvent


def get_chain(db: Session, animal_id: uuid.UUID) -> list[LedgerEvent]:
    stmt = (
        select(LedgerEvent)
        .where(LedgerEvent.animal_id == animal_id)
        .order_by(LedgerEvent.seq)
    )
    return list(db.execute(stmt).scalars().all())


def append_event(
    db: Session,
    animal_id: uuid.UUID,
    event_type: str,
    payload: dict,
    signature: str | None = None,
    signer_id: str | None = None,
) -> LedgerEvent:
    chain = get_chain(db, animal_id)
    if chain:
        seq = chain[-1].seq + 1
        prev_hash = chain[-1].hash
    else:
        seq = 0
        prev_hash = GENESIS_HASH
    event = LedgerEvent(
        animal_id=animal_id,
        seq=seq,
        event_type=event_type,
        payload=payload,
        prev_hash=prev_hash,
        hash=compute_hash(str(animal_id), seq, event_type, payload, prev_hash),
        signature=signature,
        signer_id=signer_id,
    )
    db.add(event)
    db.commit()
    db.refresh(event)
    return event


def create_animal(
    db: Session,
    *,
    breed: str,
    sex: str,
    birth_date: date,
    external_tag: str | None = None,
    sire_id: uuid.UUID | None = None,
    dam_id: uuid.UUID | None = None,
    muzzle_ref: str | None = None,
) -> Animal:
    animal = Animal(
        breed=breed,
        sex=sex,
        birth_date=birth_date,
        external_tag=external_tag,
        sire_id=sire_id,
        dam_id=dam_id,
        muzzle_ref=muzzle_ref,
    )
    db.add(animal)
    db.commit()
    db.refresh(animal)
    append_event(
        db,
        animal.id,
        "ANIMAL_REGISTERED",
        {
            "breed": breed,
            "sex": sex,
            "birth_date": birth_date.isoformat(),
            "external_tag": external_tag,
        },
    )
    return animal


def verify_chain(db: Session, animal_id: uuid.UUID) -> bool:
    expected_prev = GENESIS_HASH
    for event in get_chain(db, animal_id):
        if event.prev_hash != expected_prev:
            return False
        recomputed = compute_hash(
            str(event.animal_id), event.seq, event.event_type, event.payload, event.prev_hash
        )
        if recomputed != event.hash:
            return False
        expected_prev = event.hash
    return True
```

    return animal

def test_verify_detects_tampering(db):
    """Disable the trigger to simulate a DB-level attacker, corrupt a row,
    and confirm verify_chain catches it."""

    )

import * as SQLite from "expo-sqlite";

export type CachedAnimal = { id: string; breed: string; birth_date: string };

const db = SQLite.openDatabaseSync("truebeef.db");

export function initDb(): void {
  db.execSync(
    "CREATE TABLE IF NOT EXISTS animals (id TEXT PRIMARY KEY, breed TEXT, birth_date TEXT)"
  );
}

export function cacheAnimal(a: CachedAnimal): void {
  db.runSync(
    "INSERT OR REPLACE INTO animals (id, breed, birth_date) VALUES (?, ?, ?)",
    a.id,
    a.breed,
    a.birth_date
  );
}

export function getCachedAnimals(): CachedAnimal[] {
  return db.getAllSync<CachedAnimal>("SELECT id, breed, birth_date FROM animals");
}
